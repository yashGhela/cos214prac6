243 int TIVIDYFacade::getProgress()
244 {
245  std::vector<WorkItem*> items = getWorkItems();
246 
247  if (items.empty())
248  return 0;
249 
250  int finished = 0;
251  int active = 0;
252 
253  for (size_t i = 0; i < items.size(); ++i)
254  {
255  if (items[i]->isCancelled())
256  continue;
257 
258  ++active;
259 
260  if (items[i]->isFinished())
261  {
262  ++finished;
263  }
264  }
265 
266  if (active == 0)
267  return 0;
268 
269  return (finished * 100) / active;
270 }
